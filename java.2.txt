
import javafx.application.Application;
import javafx.geometry.Insets;
import javafx.geometry.Pos;
import javafx.scene.Scene;
import javafx.scene.control.Alert;
import javafx.scene.control.Button;
import javafx.scene.control.ComboBox;
import javafx.scene.control.Label;
import javafx.scene.control.TextField;
import javafx.scene.layout.GridPane;
import javafx.stage.Stage;

public class TipCalculator extends Application {

    @Override
    public void start(Stage stage) {

        // Create labels
        Label title = new Label("Tip Calculator");
        Label billLabel = new Label("Bill Amount (Rs.):");
        Label tipLabel = new Label("Tip Percentage:");
        Label tipResult = new Label("Tip Amount: Rs. 0.00");
        Label totalResult = new Label("Total Bill: Rs. 0.00");

        // Create text field
        TextField billField = new TextField();
        billField.setPromptText("Enter bill amount");

        // Create combo box
        ComboBox<Integer> tipBox = new ComboBox<>();
        tipBox.getItems().addAll(5, 10, 15, 20, 25);
        tipBox.setValue(15);

        // Create buttons
        Button calculateButton = new Button("Calculate");
        Button clearButton = new Button("Clear");

        // Handle Calculate button click
        calculateButton.setOnAction(event -> {
            try {
                double bill = Double.parseDouble(
                    billField.getText().trim()
                );

                if (!Double.isFinite(bill) || bill < 0) {
                    throw new IllegalArgumentException();
                }

                int percentage = tipBox.getValue();

                double tip = bill * percentage / 100.0;
                double total = bill + tip;

                tipResult.setText(
                    String.format("Tip Amount: Rs. %.2f", tip)
                );

                totalResult.setText(
                    String.format("Total Bill: Rs. %.2f", total)
                );

            } catch (IllegalArgumentException e) {
                Alert alert = new Alert(Alert.AlertType.ERROR);
                alert.setTitle("Invalid Input");
                alert.setHeaderText("Invalid bill amount");
                alert.setContentText(
                    "Please enter a valid, non-negative number."
                );
                alert.showAndWait();
            }
        });

        // Handle Clear button click
        clearButton.setOnAction(event -> {
            billField.clear();
            tipBox.setValue(15);
            tipResult.setText("Tip Amount: Rs. 0.00");
            totalResult.setText("Total Bill: Rs. 0.00");
        });

        // Style title
        title.setStyle(
            "-fx-font-size: 24px; -fx-font-weight: bold;"
        );

        // Create layout
        GridPane grid = new GridPane();
        grid.setPadding(new Insets(20));
        grid.setHgap(15);
        grid.setVgap(15);
        grid.setAlignment(Pos.CENTER);

        // Add components to layout
        grid.add(title, 0, 0, 2, 1);
        grid.add(billLabel, 0, 1);
        grid.add(billField, 1, 1);
        grid.add(tipLabel, 0, 2);
        grid.add(tipBox, 1, 2);
        grid.add(calculateButton, 0, 3);
        grid.add(clearButton, 1, 3);
        grid.add(tipResult, 0, 4, 2, 1);
        grid.add(totalResult, 0, 5, 2, 1);

        // Create scene
        Scene scene = new Scene(grid, 420, 320);

        // Configure stage
        stage.setTitle("JavaFX Tip Calculator");
        stage.setScene(scene);
        stage.setResizable(false);
        stage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
